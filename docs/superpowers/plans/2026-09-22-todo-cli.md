```rust
use rusqlite::Connection;

fn open_and_migrate(path: &std::path::Path) -> anyhow::Result<Connection> {
    let conn = Connection::open(path).context("opening database")?;
    conn.execute_batch(
        r#"
        CREATE TABLE IF NOT EXISTS todos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            title TEXT NOT NULL,
            done INTEGER NOT NULL DEFAULT 0,
            priority TEXT NOT NULL DEFAULT 'medium',
            due DATE,
            tags TEXT NOT NULL DEFAULT '',
            created_at TEXT NOT NULL,
            completed_at TEXT,
            deleted_at TEXT
        );
        CREATE INDEX IF NOT EXISTS idx_done ON todos (done);
        CREATE INDEX IF NOT EXISTS idx_priority ON todos (priority);
        CREATE INDEX IF NOT EXISTS idx_deleted_at ON todos (deleted_at);
        "#,
    )
    .context("running migrations")?;
    Ok(conn)
}
```

> **`execute_batch`** runs multiple SQL statements separated by `;`. The `r#"..."#` is a raw string literal — no escaping needed for `"` inside.

> **`open_in_memory`** creates a SQLite DB that lives only in RAM — perfect for tests.

```rust
use thiserror::Error;

#[derive(Error, Debug)]
pub enum AppError {
    #[error("todo #{0} not found")]
    NotFound(i64),

    #[error("todo #{0} is already done")]
    AlreadyDone(i64),

    #[error("todo #{0} is not deleted")]
    NotDeleted(i64),

    #[error("invalid priority: {0} (expected low|medium|high)")]
    InvalidPriority(String),

    #[error("invalid date: {0} (expected YYYY-MM-DD)")]
    InvalidDate(String),

    #[error("database error: {0}")]
    Db(#[from] rusqlite::Error),

    #[error("io error: {0}")]
    Io(#[from] std::io::Error),
}

pub type Result<T> = std::result::Result<T, AppError>;
```

> **`#[from]` on a field** auto-generates `From<rusqlite::Error> for AppError` — that's why `?` can convert DB errors without manual code.

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Priority {
    Low,
    Medium,
    High,
}

impl Priority {
    pub fn parse(s: &str) -> Result<Self> {
        match s {
            "low" => Ok(Priority::Low),
            "medium" => Ok(Priority::Medium),
            "high" => Ok(Priority::High),
            _ => Err(AppError::InvalidPriority(s.to_string())),
        }
    }

    pub fn as_str(&self) -> &'static str {
        match self {
            Priority::Low => "low",
            Priority::Medium => "medium",
            Priority::High => "high",
        }
    }
}
```

```rust
use crate::cli::AddArgs;
use crate::error::{AppError, Result};
use crate::models::Priority;
use chrono::Utc;
use rusqlite::Connection;

pub fn run(conn: &Connection, args: &AddArgs) -> Result<String> {
    let priority = Priority::parse(&args.priority)?;
    let due = args
        .due
        .as_deref()
        .map(|s| chrono::NaiveDate::parse_from_str(s, "%Y-%m-%d"))
        .transpose()
        .map_err(|_| AppError::InvalidDate(args.due.clone().unwrap_or_default()))?;
    let now = Utc::now().naive_utc();
    conn.execute(
        "INSERT INTO todos (title, priority, due, tags, created_at) VALUES (?1, ?2, ?3, ?4, ?5)",
        rusqlite::params![args.title, priority.as_str(), due, args.tags.join(","), now],
    )?;
    let id = conn.last_insert_rowid();
    Ok(format!("Added #{}: {}", id, args.title))
}
```

> **No `tags_to_csv` helper needed:** `[String]::join(",")` works directly — std implements `join` on slices whose elements are `String` or `&str`.
>
> **`params![...]` accepts chrono types** because of the `chrono` feature added in Step 0. Without it: compile error `the trait bound Option<NaiveDate>: ToSql is not satisfied`.

```rust
use crate::cli::ListArgs;
use crate::error::Result;
use rusqlite::Connection;

pub fn run(conn: &Connection, args: &ListArgs) -> Result<String> {
    let mut sql = String::from("SELECT id, title, done, priority, due, tags FROM todos WHERE deleted_at IS NULL");
    let mut wheres: Vec<String> = vec![];
    let mut binds: Vec<String> = vec![];

    if !args.all {
        wheres.push("done = 0".to_string());
    }
    if let Some(t) = &args.tag {
        wheres.push("',' || tags || ',' LIKE ?".to_string());
        binds.push(format!(",{},", t));
    }
    if let Some(p) = &args.priority {
        wheres.push("priority = ?".to_string());
        binds.push(p.clone());
    }
    if let Some(s) = &args.search {
        wheres.push("title LIKE ?".to_string());
        binds.push(format!("%{}%", s));
    }
    if args.trash {
        sql = sql.replace("WHERE deleted_at IS NULL", "WHERE deleted_at IS NOT NULL");
        wheres.clear();
        binds.clear();
    }
    if !wheres.is_empty() {
        sql.push_str(" AND ");
        sql.push_str(&wheres.join(" AND "));
    }
    sql.push_str(" ORDER BY id ASC");

    let mut stmt = conn.prepare(&sql)?;
    let rows = stmt.query_map(rusqlite::params_from_iter(binds.iter()), |row| {
        Ok((
            row.get::<_, i64>(0)?,
            row.get::<_, String>(1)?,
            row.get::<_, i64>(2)?,
            row.get::<_, String>(3)?,
            row.get::<_, Option<String>>(4)?,
            row.get::<_, String>(5)?,
        ))
    })?;

    let mut out = String::new();
    for row in rows {
        let (id, title, done, priority, due, tags) = row?;
        let mark = if done == 1 { "[x]" } else { "[ ]" };
        let due_s = due.unwrap_or_else(|| "-".into());
        let tag_s = if tags.is_empty() { "-".into() } else { tags };
        out.push_str(&format!("{} {} {:>5} {:<10} {} ({})\n", id, mark, priority, due_s, title, tag_s));
    }
    Ok(out)
}
```

> **`String` accumulation vs `println!`:** we build a single String and return it. `main` prints. Keeps the handler pure and easy to test.

> **`params_from_iter`** lets you pass a `Vec<String>` (or any iterator) to `query_map`. The `?` placeholders bind positionally.

```rust
    if let Some(p) = &args.priority {
        let prio = Priority::parse(p)?;
        sets.push("priority = ?".into());
        binds.push(prio.as_str().to_string());

            .map_err(|_| AppError::InvalidDate(due.clone()))?;
        sets.push("due = ?".into());
        binds.push(due.clone());
    }
    if let Some(tags) = &args.tags {
        sets.push("tags = ?".into());
        binds.push(tags.join(","));
    }

    if sets.is_empty() {
        return Ok(format!("Nothing to update for #{}", args.id));
    }

    let sql = format!(
        "UPDATE todos SET {} WHERE id = ? AND deleted_at IS NULL",
        sets.join(", ")
    );
    let id_str = args.id.to_string();
    let mut all_binds = binds;
    all_binds.push(id_str);

    let updated = conn.execute(&sql, rusqlite::params_from_iter(all_binds.iter()))?;
    if updated == 0 {
        return Err(AppError::NotFound(args.id));
    }
    Ok(format!("Updated #{}", args.id))
}
```

```rust
use chrono::Utc;
use rusqlite::Connection;

const TRASH_LIMIT: i64 = 15;

pub fn run(conn: &Connection, id: i64) -> Result<String> {
    let now = Utc::now().naive_utc();
    let updated = conn.execute(
        "UPDATE todos SET deleted_at = ?1 WHERE id = ?2 AND deleted_at IS NULL",
        rusqlite::params![now, id],
    )?;
    if updated == 0 {
        return Err(AppError::NotDeleted(id));
    }

    let count: i64 = conn.query_row(
        "SELECT COUNT(*) FROM todos WHERE deleted_at IS NOT NULL",
        [],
        |r| r.get(0),
    )?;

    if count > TRASH_LIMIT {
        let excess = count - TRASH_LIMIT;
        conn.execute(
            "DELETE FROM todos WHERE id IN (
                SELECT id FROM todos WHERE deleted_at IS NOT NULL
                ORDER BY deleted_at ASC, id ASC LIMIT ?1
            )",
            [excess],
        )?;
    }

    Ok(format!("Deleted #{} (trash buffer: {} max)", id, TRASH_LIMIT))
}
```

```rust
        "UPDATE todos SET deleted_at = NULL WHERE id = ?1 AND deleted_at IS NOT NULL",
        [id],
    )?;
    if updated == 0 {
        return Err(AppError::NotFound(id));
    }
    Ok(format!("Restored #{}", id))
}
```

```rust
/// Binary name = package name = "todo-cli", NOT the clap display name "todo".

/// Unique-per-run title so reruns don't collide with rows from a previous test run.
fn uniq(base: &str) -> String {
    format!("{} {}", base, std::process::id())
}

#[test]

    bin()
        .arg("--help")
        .assert()
        .success()
        .stdout(predicate::str::contains("Usage"));
}

#[test]
fn add_then_list_shows_todo() {
    let title = uniq("integration test");
    bin().arg("add").arg(&title).assert().success();

    bin()
        .arg("list")
        .assert()
        .success()
        .stdout(predicate::str::contains(title));
}

#[test]
fn delete_hides_from_list() {
    let title = uniq("to be deleted");
    bin().arg("add").arg(&title).assert().success();

    let output = bin().arg("list").output().unwrap();
    let stdout = String::from_utf8(output.stdout).unwrap();
    let id_line = stdout
        .lines()
        .find(|l| l.contains(&title))
        .expect("todo not in list");
    let id: i64 = id_line
        .split_whitespace()
        .next()
        .unwrap()
        .parse()
        .unwrap();

    bin().arg("delete").arg(id.to_string()).assert().success();

    bin()
        .arg("list")
        .assert()
        .stdout(predicate::str::contains(title.as_str()).not());
}
```

> **Two traps fixed here:**
> 1. `cargo_bin` takes the **binary** name, which equals the package name (`todo-cli`) — not the clap display name (`todo`). Wrong name = panic at `.unwrap()`.
> 2. Integration tests hit the real user-data-dir DB and persist across runs. Without unique titles, the second run finds leftover rows and fails. `std::process::id()` changes each run.
